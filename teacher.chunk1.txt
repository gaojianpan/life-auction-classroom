function tab(name){
  ["create","open","history"].forEach(x=>$("pane-"+x).classList.toggle("hidden",x!==name));
  document.querySelectorAll(".tab").forEach(b=>b.classList.toggle("active",b.dataset.tab===name));
}
function esc(s){return String(s||"").replace(/[&<>"']/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]))}
function secLeft(end){return Math.max(0,Math.ceil((end-Date.now())/1000))}
function seconds(){return Math.max(5,Number($("seconds").value)||12)}
function priceFor(index){
  const raw=priceDrafts[index];
  if(raw===undefined||raw===null||String(raw).trim()==="")return 100;
  const n=Math.floor(Number(raw));
  return Number.isFinite(n)?Math.max(0,Math.min(1000,n)):100;
}

function enterControl(code,title,state){
  currentCode=code;S=state;$("control").classList.remove("hidden");
  $("controlTitle").textContent=title;$("controlCode").textContent=code;
  $("joinUrl").value=location.origin+"/?code="+code;$("qr").src="/api/qr/"+code+".png";
  window.scrollTo({top:$("control").offsetTop-10,behavior:"smooth"});renderControl()
}

function renderControl(){
  if(!S)return;
  $("controlTitle").textContent=S.title;$("controlCode").textContent=S.code;
  $("studentCount").textContent=S.students.length;
  $("onlineCount").textContent=S.students.filter(x=>x.connected).length;
  $("done").textContent=`${S.history.length} / 15`;
  const a=S.currentAuction;
  $("current").innerHTML=a?`<b>${a.item.icon} ${a.item.title}</b><br><span class="muted">${esc(a.item.desc)}</span><br><span class="amber">当前统一成交价：<b>${a.cutoff}</b></span>`
    :`当前阶段：<b>${({lobby:"待机",auction:"轮间等待",exchange:"交换市场",reflection:"课堂讨论",ended:"已归档"})[S.mode]||S.mode}</b>`;
  $("bids").innerHTML=a&&a.bids.length?a.bids.map(b=>`<div class="bidrow ${b.winning?'win':''}"><div>#${b.rank}</div><div><b>${esc(b.name)}</b>${b.studentNo?`<div class="muted small">学号 ${esc(b.studentNo)}</div>`:""}</div><div>${b.amount}${b.winning?`<div class="green small">按此价成交</div>`:""}</div></div>`).join("")
    :`<p class="muted">等待竞价</p>`;
  $("studentRoster").innerHTML=S.students.length?S.students
    .slice().sort((a,b)=>String(a.studentNo||"").localeCompare(String(b.studentNo||""),"zh-CN",{numeric:true}))
    .map(s=>`<div class="listing"><div style="min-width:120px"><b>${esc(s.studentNo||"未填学号")}</b><div>${esc(s.name)}</div></div><div class="${s.connected?'green':'muted'}">${s.connected?'● 在线':'○ 离线'}</div><div><span class="muted small">余额</span><br><b>${s.coins}</b></div><div style="flex:1"><span class="muted small">已拍得模块</span><br>${s.inventory.length?s.inventory.map(i=>`${i.icon} ${esc(i.customText||i.title)}`).join("、"):`<span class="muted">暂无</span>`}</div></div>`).join("")
    :`<p class="muted">尚无学生加入。</p>`;
  $("items").innerHTML=S.items.map((it,i)=>`<div class="stat"><div style="font-size:30px">${it.icon}</div><b>${it.id}. ${it.title}</b><div class="muted small" style="min-height:42px;margin:7px 0">${it.hook}</div><div class="muted small" style="margin-bottom:4px">起拍价（留空默认100）</div><input id="base_${i}" inputmode="numeric" min="0" max="1000" placeholder="100" value="${esc(priceDrafts[i]??"")}" oninput="priceDrafts[${i}]=this.value" style="margin-bottom:8px"><button class="secondary" onclick="startAt(${i})">开拍</button></div>`).join("");
}
