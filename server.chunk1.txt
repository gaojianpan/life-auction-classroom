app.get("/api/qr/:code.png", async (req, res) => {
  const code = normalizeCode(req.params.code);
  if (!code) return res.status(400).send("bad code");
  const proto = req.get("x-forwarded-proto") || req.protocol || "https";
  const host = req.get("host");
  const url = `${proto}://${host}/?code=${code}`;
  try {
    const png = await QRCode.toBuffer(url, { type: "png", width: 480, margin: 2 });
    res.type("png").send(png);
  } catch {
    res.status(500).send("QR generation failed");
  }
});

function normalizeCode(v) {
  const s = String(v || "").replace(/\D/g, "").slice(0, 6);
  return s.length === 6 ? s : "";
}
function cleanText(v, n=40) {
  return String(v || "").trim().replace(/\s+/g, " ").slice(0, n);
}
function id() { return crypto.randomBytes(12).toString("hex"); }

function hashPin(pin) {
  const salt = crypto.randomBytes(16).toString("hex");
  const hash = crypto.scryptSync(String(pin), salt, 64).toString("hex");
  return `${salt}:${hash}`;
}
function verifyPin(pin, stored) {
  try {
    const [salt, expected] = String(stored).split(":");
    const actual = crypto.scryptSync(String(pin), salt, 64);
    return crypto.timingSafeEqual(actual, Buffer.from(expected, "hex"));
  } catch { return false; }
}

async function uniqueCode() {
  for (let i=0;i<50;i++) {
    const code = String(crypto.randomInt(100000, 1000000));
    const r = await pool.query("SELECT 1 FROM classrooms WHERE code=$1", [code]);
    if (!r.rowCount) return code;
  }
  throw new Error("Unable to allocate classroom code");
}
