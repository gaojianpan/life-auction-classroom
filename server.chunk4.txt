      const hashed = hashPin(teacherPin);
      const r = await pool.query(
        `INSERT INTO classrooms(code,title,teacher_pin_hash,state)
         VALUES($1,$2,$3,$4::jsonb)
         RETURNING id,created_at`,
        [code,title,hashed,JSON.stringify(state)]
      );
      const room = {
        id:r.rows[0].id, code, title, teacherPinHash:hashed, state,
        persistChain:Promise.resolve()
      };
      rooms.set(code, room);
      socket.data.teacherCode = code;
      socket.join(roomChannel(code));
      cb?.({ok:true,code,title,state:snapshot(room)});
    } catch (e) {
      console.error(e); cb?.({ok:false,error:"创建课堂失败"});
    }
  });

  socket.on("openTeacherClassroom", async ({code,pin}, cb) => {
    try {
      const room = await getRoom(code);
      if (!room) return cb?.({ok:false,error:"课堂码不存在"});
      if (!verifyPin(cleanText(pin,20), room.teacherPinHash))
        return cb?.({ok:false,error:"课堂教师口令错误"});
      socket.data.teacherCode = room.code;
      socket.join(roomChannel(room.code));
      cb?.({ok:true,code:room.code,title:room.title,state:snapshot(room)});
      socket.emit("roomState", snapshot(room));
    } catch {
      cb?.({ok:false,error:"进入课堂失败"});
    }
  });

  socket.on("joinClassroom", async ({code,name,studentNo,clientKey}, cb) => {
    try {
      const room = await getRoom(code);
      if (!room) return cb?.({ok:false,error:"课堂码不存在"});
      if (room.state.mode === "ended") return cb?.({ok:false,error:"该课堂已结束归档"});
      name = cleanText(name,12);
      studentNo = cleanText(studentNo,24).replace(/\s+/g, "");
      clientKey = cleanText(clientKey,80);
      if (!name) return cb?.({ok:false,error:"请输入课堂昵称 / 姓名"});
      if (!studentNo) return cb?.({ok:false,error:"请输入学号"});
      if (!/^[A-Za-z0-9_-]+$/.test(studentNo))
        return cb?.({ok:false,error:"学号仅支持字母、数字、下划线和短横线"});
      if (!clientKey) return cb?.({ok:false,error:"浏览器身份生成失败，请刷新重试"});

      let s = Object.values(room.state.students).find(x=>x.clientKey===clientKey);
      const duplicate = Object.values(room.state.students).find(x=>x.studentNo===studentNo && (!s || x.id!==s.id));
      if (duplicate) return cb?.({ok:false,error:"该学号已加入本课堂，请检查学号；如需换设备请联系教师处理"});

      if (!s) {
        s = {
          id:id(), clientKey, name, studentNo, coins:1000, inventory:[],
          connected:true, joinedAt:Date.now()
        };
        room.state.students[s.id] = s;
      } else {
        s.name = name; s.studentNo = studentNo; s.connected = true;
      }
      socket.data.studentId = s.id;
      socket.data.studentCode = room.code;
      socket.join(roomChannel(room.code));
      await queuePersist(room);
      cb?.({ok:true,studentId:s.id,studentNo:s.studentNo,code:room.code,title:room.title});
      emitRoom(room);
    } catch (e) {
      console.error(e); cb?.({ok:false,error:"加入课堂失败"});
    }
  });

  });
