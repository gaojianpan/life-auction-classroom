;
  return room;
}

function queuePersist(room) {
  const snapshot = JSON.stringify(room.state);
  const ended = room.state.endedAt ? new Date(room.state.endedAt) : null;
  room.persistChain = room.persistChain
    .catch(()=>{})
    .then(() => pool.query(
      "UPDATE classrooms SET state=$1::jsonb,updated_at=NOW(),ended_at=$2 WHERE id=$3",
      [snapshot, ended, room.id]
    ));
  return room.persistChain;
}

function studentPublic(s) {
  return {
    id:s.id, name:s.name, studentNo:s.studentNo || "", coins:s.coins, inventory:s.inventory,
    connected:!!s.connected, joinedAt:s.joinedAt
  };
}

function auctionView(room) {
  const a = room.state.currentAuction;
  if (!a) return null;
  const sorted = Object.values(a.bids || {}).sort((x,y)=>y.amount-x.amount || x.ts-y.ts);
  const highest = sorted.length ? sorted[0].amount : a.item.base;
  return {
    item:a.item,
    timerEnd:room.state.timerEnd,
    bids:sorted.map((b,i)=>({
      studentId:b.studentId, name:b.name, studentNo:b.studentNo || "", amount:b.amount,
      rank:i+1, winning:b.amount===highest
    })),
    cutoff:highest
  };
}

function snapshot(room) {
  return {
    code:room.code,
    title:room.title,
    mode:room.state.mode,
    items:ITEMS,
    currentIndex:room.state.currentIndex,
    currentAuction:auctionView(room),
    timerEnd:room.state.timerEnd,
    exchangeEnd:room.state.exchangeEnd,
    students:Object.values(room.state.students).map(studentPublic),
    history:room.state.history,
    listings:room.state.listings,
    endedAt:room.state.endedAt
  };
}

function emitRoom(room) {
  io.to(roomChannel(room.code)).emit("roomState", snapshot(room));
}

async function startItem(room, index, seconds=12, basePrice=100) {
  if (room.state.mode === "ended") throw new Error("课堂已归档");
  index = Number(index);
  if (index < 0 || index >= ITEMS.length) throw new Error("拍卖品不存在");
  if (room.state.currentAuction) await closeAuction(room);
  basePrice = Math.floor(Number(basePrice));
  if (!Number.isFinite(basePrice) || basePrice < 0 || basePrice > 1000) basePrice = 100;
  room.state.mode = "auction";
  room.state.currentIndex = index;
  room.state.currentAuction = { item: {...ITEMS[index], base:basePrice}, bids: {} };
  room.state.timerEnd = Date.now() + Math.max(5, Math.min(120, Number(seconds)||12))*1000;
  await queuePersist(room);
  emitRoom(room);
}

async function closeAuction(room) {
  const a = room.state.currentAuction;
  if (!a) return;
  const sorted = Object.values(a.bids || {}).sort((x,y)=>y.amount-x.amount || x.ts-y.ts);
  const clearingPrice = sorted.length ? sorted[0].amount : null;
  const winners = clearingPrice === null ? [] : sorted.filter(x=>x.amount===clearingPrice);
  const settled = [];

  for (const w of winners) {
    const s = room.state.students[w.studentId];
    if (!s || s.coins < clearingPrice) continue;
    s.coins -= clearingPrice;
    const copy = {
      copyId:id(), itemId:a.item.id, title:a.item.title, icon:a.item.icon,
      paid:clearingPrice, customText:w.customText || ""
    };
    s.inventory.push(copy);
    settled.push({
      studentId:s.id, name:s.name, studentNo:s.studentNo || "", amount:clearingPrice,
      copyId:copy.copyId, customText:copy.customText
    });
  }

  room.state.history.push({
    round:room.state.history.length+1,
    itemId:a.item.id, title:a.item.title, icon:a.item.icon,
    clearingPrice:clearingPrice,
    winners:settled, closedAt:Date.now()
  });
  room.state.currentAuction = null;
  room.state.timerEnd = null;
  await queuePersist(room);
  emitRoom(room);
}

async function reconcile(room) {
  const now = Date.now();
  if (room.state.currentAuction && room.state.timerEnd && now >= room.state.timerEnd) {
    await closeAuction(room);
  }
  if (room.state.mode === "exchange" && room.state.exchangeEnd && now >= room.state.exchangeEnd) {
    room.state.mode = "reflection";
    room.state.exchangeEnd = null;
    await queuePersist(room);
  }
}
